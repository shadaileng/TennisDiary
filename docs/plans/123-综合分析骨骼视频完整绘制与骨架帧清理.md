# 综合分析骨骼视频完整绘制与骨架帧清理

## 1. 背景

分析报告 #9（综合分析，mode=full）的骨骼视频仅 8 帧，而拼接后视频有 210 帧（7秒）。

| 分析报告 | 模式 | 拼接视频 | 骨骼帧数 | 骨骼视频 |
|---|---|---|---|---|
| #3 (single) | single | 1段 | 61帧 | 正常 |
| #9 (full) | full | 多段拼接7s/210帧 | **仅8帧** | 异常 |

**根因链**：

1. `pipeline._compute_pose()` 调用 `pose_service.analyze_frames()` 未传 `full_frames` 参数
2. `_should_use_full_frames()` 按 `duration(7s) > threshold(3s)` 判定为抽样模式
3. 仅处理 8 帧 → 骨骼视频仅 8 帧 → 帧率 = 8/7 ≈ 1.14fps，几乎静止

**附加问题**：骨架帧图片（`sk*.jpg`）分析完成后未清理，占用磁盘且无用途。

## 2. 目标

1. 综合分析（mode=full）的骨骼视频逐帧绘制，帧数与原视频一致
2. 分析完成后清理骨架帧图片（`sk*.jpg`），仅保留骨架视频和缩略图
3. 缩略图（`thumb`）独立于骨架帧，不因清理而丢失
4. 删除分析时兜底检查残留骨架帧文件，确保无孤立文件

```python
def _compute_pose(self, frame_urls: list[str], video_result: dict, metadata: dict) -> dict:
    """姿态推理（线程内执行，纯计算无 DB）"""
    t0 = time.time()
    from app.services import file_service, pose_service

    save_skeleton = True
    duration = video_result.get("duration")
    frame_rate = video_result.get("frame_rate")
    working_path = video_result.get("working_path")
    video_url = file_service.abs_path_to_rel(working_path) if working_path else None

    # full 模式强制逐帧生成骨架视频，single 模式按阈值自动判断
    full_frames = True if metadata.get("mode") == "full" else None

    result = pose_service.analyze_frames(
        frames=None,
        video_url=video_url,
        save_skeleton=save_skeleton,
        duration=duration,
        frame_rate=frame_rate,
        frame_urls=frame_urls,
        full_frames=full_frames,  # 新增
    )

    log.info(f"管线-姿态推理耗时: {time.time() - t0:.2f}s frames={len(frame_urls)}")
    return result
```

```python
def _write_pose_result(self, pose_result: dict) -> None:
    """写入姿态结果到数据库"""
    if not (self.analysis_id and pose_result):
        return

    from app.services import file_service

    skeleton_video = pose_result.get("skeleton_video_info")
    skeleton_thumb = pose_result.get("skeleton_thumb_info")
    user_id = self._get_user_id()

    # 仅登记骨架视频和缩略图，骨架帧不登记（分析完成后清理）
    all_files = []
    if skeleton_video:
        all_files.append(skeleton_video)
    if skeleton_thumb:
        all_files.append(skeleton_thumb)

    if all_files:
        try:
            file_service.batch_get_or_create_files(
                db=self.db,
                user_id=user_id,
                files=all_files,
                business_type="analysis",
                business_id=self.analysis_id,
            )
        except Exception as exc:  # noqa: BLE001
            log.warning("文件批量登记失败(非致命): %s - %s", type(exc).__name__, str(exc)[:120])

    # pose_for_db 中 skeleton_frames 改为空列表（帧已清理）
    pose_for_db = {
        "frames": pose_result.get("frames"),
        "metrics": pose_result.get("metrics"),
        "detected": pose_result.get("detected"),
        "skeleton_frames": [],  # 骨架帧已清理，不落库
        "skeleton_video_url": pose_result.get("skeleton_video_url"),
    }
    stmt = (
        sa_update(Analysis)
        .where(Analysis.id == self.analysis_id)
        .values(
            pose=json.dumps(pose_for_db, ensure_ascii=False),
            thumb=skeleton_thumb["rel_path"] if skeleton_thumb else None,
        )
    )
    self.db.execute(stmt)
    self.db.commit()
```

```python
def _finalize(self) -> None:
    """收尾更新状态"""
    t0 = time.time()
    analysis = self.db.query(Analysis).filter(Analysis.id == self.analysis_id).first()
    if analysis and analysis.video_url:
        self._update_analysis_field(status="completed")
        pipeline_status = _parse_pipeline_status(analysis.pipeline_status)
        if pipeline_status:
            pipeline_status["step"] = PipelineStep.FINALIZE.value
            pipeline_status["progress"] = 100
            pipeline_status["steps"][PipelineStep.FINALIZE.value] = {
                "status": StepStatus.COMPLETED.value,
                "ts": time.time(),
                "duration_s": round(time.time() - t0, 2),
            }
            self._update_pipeline_status(pipeline_status)

        # 清理采样帧和骨架帧
        self._cleanup_sampled_frames(analysis.video_url)
        self._cleanup_skeleton_frames(analysis.video_url)
    else:
        log.warning(
            "分析记录 video_url 为空，保留 processing 孤儿",
            analysis_id=self.analysis_id,
        )
```

#### D. 新增 `_cleanup_skeleton_frames()` 方法

```python
def _cleanup_skeleton_frames(self, video_url: str) -> None:
    """删除骨架帧文件（_sk*.jpg），保留骨架视频和缩略图"""
    import glob

    from app.services import file_service

    video_dir = os.path.dirname(file_service.rel_path_to_abs(video_url))
    base = os.path.splitext(os.path.basename(video_url))[0]
    pattern = os.path.join(video_dir, f"{base}_sk*.jpg")
    cleaned = 0
    for path in glob.glob(pattern):
        try:
            os.remove(path)
            cleaned += 1
        except OSError as exc:
            log.warning("删除骨架帧失败: %s - %s", path, exc)
    if cleaned:
        log.info("已清理骨架帧: {} 个文件 business=analysis/{}", cleaned, self.analysis_id)
```

```python
def _extract_first_frame(video_path: str, out_path: str) -> bool:
    """从视频提取首帧保存为 JPEG，返回是否成功"""
    ffmpeg = find_ffmpeg()
    if not ffmpeg:
        return False
    proc = subprocess.run(
        [ffmpeg, "-y", "-i", video_path, "-vframes", "1", "-q:v", "2", out_path],
        capture_output=True,
        timeout=30,
    )
    return proc.returncode == 0 and os.path.isfile(out_path)
```

#### B. 新增 `_cleanup_orphan_intermediate_frames()` 兜底清理

```python
def _cleanup_orphan_intermediate_frames(db: Session, analysis) -> int:
    """删除分析时兜底检查：清理残留的抽样帧和骨架帧图片，防止孤立文件

    正常流程中抽样帧和骨架帧已在 _finalize() 清理，此函数处理异常残留。
    返回清理的文件数量。
    """
    import glob

    count = 0
    if not analysis.video_url:
        return 0
    try:
        video_dir = os.path.dirname(rel_path_to_abs(analysis.video_url))
        base = os.path.splitext(os.path.basename(analysis.video_url))[0]
        # 清理抽样帧（_f*.jpg）和骨架帧（_sk*.jpg）
        for pattern in [f"{base}_f*.jpg", f"{base}_sk*.jpg"]:
            for path in glob.glob(os.path.join(video_dir, pattern)):
                try:
                    os.remove(path)
                    count += 1
                except OSError as exc:
                    log.warning("删除残留中间帧失败: %s - %s", path, exc)
    except Exception as exc:  # noqa: BLE001 - 兜底清理不应抛错
        log.warning("兜底清理中间帧异常: %s - %s", type(exc).__name__, str(exc)[:120])
    if count:
        log.info("删除分析兜底清理中间帧: {} 个文件 analysis_id={}", count, analysis.id)
    return count
```

## 6. 风险与注意事项

| 风险 | 影响 | 缓解 |

| full 模式逐帧推理耗时增加 | 7秒视频约 210 帧，MediaPipe 推理约 30-60s | 用户明确选择综合分析，可接受；阈值可通过配置调整 |
